# SEC-20 Flow view Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [x]`) syntax for tracking.

**Goal:** One store's captured pages in shopping order (home → collection → product → cart → checkout) as a strip on desktop and a stack on phones, plus an index that counts stores with complete flows.

**Architecture:** A pure module (`lib/flows.ts`) turns a store's `ProfilePage[]` into five fixed steps and counts completeness across the block index in one pass. Two client components render it: the flow page and the index. Data comes from the existing loaders (`loadSiteView`, `loadBlockIndex`); nothing new is fetched.

**Tech Stack:** Next.js 16 app router, React 19, Tailwind 4 (with the `touch:` variant from SEC-22), Node 24 type stripping for the check script.

Spec: `docs/superpowers/specs/2026-09-24-flow-view-design.md`.

## Global Constraints

- Complete = home, collection and product captured. Cart optional. Checkout always a gap with the copy "Not captured: we never capture checkout".
- Real captures only: `isMock(block)` blocks never count.
- Step order is fixed: `["home", "collection", "product", "cart", "checkout"]`.
- Relative `.ts` imports inside `web/src/lib` (so `node scripts/flow-check.ts` runs); `@/` aliases only in components and routes.
- 44px targets under `touch:`, `focus-visible:ring-2 focus-visible:ring-ring`, `text-link` for accent text.
- Store-specific pages never show mock screenshots.

```ts
// A store's flow (SEC-20): its captured pages in the order a shopper walks a
// store. Pure: no React, no DOM, no fetch.
import type { Block } from "@/contracts/block";
import type { Store } from "@/contracts/store";
import type { BlockIndex } from "./block-source.ts";
import { isMock } from "./mock-blocks.ts";
import { pagesFromBlocks, storeFromBlocks, type ProfilePage } from "./site-profile.ts";

export const FLOW_STEPS = ["home", "collection", "product", "cart", "checkout"] as const;
export type FlowStepType = (typeof FLOW_STEPS)[number];
export const REQUIRED_STEPS: readonly FlowStepType[] = ["home", "collection", "product"];

export type FlowStep = { type: FlowStepType; pages: ProfilePage[]; gap: "never" | "missing" | null };
export type Flow = { steps: FlowStep[]; complete: boolean; missing: FlowStepType[]; captured: number };
export type FlowSummary = { store: Store; listed: boolean; flow: Flow };

const hasContent = (p: ProfilePage) => Boolean(p.desktop || p.mobile || p.desktopBlocks.length || p.mobileBlocks.length);

export function flowOf(pages: ProfilePage[]): Flow {
  const steps = FLOW_STEPS.map((type): FlowStep => {
    const list = pages.filter((p) => p.type === type && hasContent(p)).sort((a, b) => a.url.localeCompare(b.url));
    return { type, pages: list, gap: list.length ? null : type === "checkout" ? "never" : "missing" };
  });
  const missing = REQUIRED_STEPS.filter((t) => steps.find((s) => s.type === t)!.pages.length === 0);
  return { steps, complete: missing.length === 0, missing, captured: steps.filter((s) => s.pages.length).length };
}

export function flowSummaries(index: BlockIndex): FlowSummary[] {
  const byHost = new Map<string, Block[]>();
  for (const b of index.blocks) {
    if (isMock(b)) continue;
    const list = byHost.get(b.host);
    if (list) list.push(b);
    else byHost.set(b.host, [b]);
  }
  const out: FlowSummary[] = [];
  for (const [host, blocks] of byHost) {
    const listed = index.storeByHost.get(host);
    out.push({ store: listed ?? storeFromBlocks(host, blocks), listed: Boolean(listed), flow: flowOf(pagesFromBlocks(blocks)) });
  }
  const rank = (s: FlowSummary) => s.store.rank ?? Number.POSITIVE_INFINITY;
  return out.sort((a, b) => Number(b.flow.complete) - Number(a.flow.complete) || b.flow.captured - a.flow.captured || rank(a) - rank(b) || a.store.brand.localeCompare(b.store.brand));
}

export function flowCounts(list: FlowSummary[]) {
  return {
    complete: list.filter((s) => s.flow.complete).length,
    withCart: list.filter((s) => s.flow.complete && s.flow.steps[3].pages.length > 0).length,
    captured: list.length,
  };
}
```
